اسکریپت دیتابیس باشگاه بدنسازی

کد SQL ارسالی برای ساخت جدول‌ها، توابع، سیاست‌های امنیتی و حرکات نمونه. برای کپی کردن، دکمه زیر را بزن.

-- ========== پاک‌سازی نسخه‌های قبلی ==========
drop table if exists memberships, plan_items, sets, workouts, exercises, profiles cascade;
drop function if exists handle_new_user() cascade;
drop function if exists is_my_student(uuid) cascade;
drop function if exists is_coach(uuid) cascade;

-- ========== جدول‌ها ==========
create table profiles (
  id uuid primary key references auth.users on delete cascade,
  username text unique not null,
  full_name text not null default '',
  role text not null check (role in ('coach', 'student', 'self')),
  coach_id uuid references profiles(id),
  invite_code text unique,
  created_at timestamptz default now()
);

create table exercises (
  id bigint generated always as identity primary key,
  name text not null,
  muscle_group text,
  description text,
  equipment text check (equipment in ('dumbbell', 'barbell', 'machine')),
  created_by uuid references profiles(id)
);

create table workouts (
  id bigint generated always as identity primary key,
  student_id uuid not null references profiles(id) on delete cascade,
  workout_date date not null default current_date,
  notes text
);

create table sets (
  id bigint generated always as identity primary key,
  workout_id bigint not null references workouts(id) on delete cascade,
  exercise_id bigint not null references exercises(id),
  set_number int not null,
  weight_kg numeric(5,1) not null,
  reps int not null
);

create table plan_items (
  id bigint generated always as identity primary key,
  student_id uuid not null references profiles(id) on delete cascade,
  day_index int not null check (day_index between 0 and 6),
  exercise_id bigint not null references exercises(id),
  sets int not null default 3,
  reps text not null default '10',
  note text,
  position int not null default 0,
  created_at timestamptz default now()
);

create table memberships (
  student_id uuid primary key references profiles(id) on delete cascade,
  total_sessions int not null default 0 check (total_sessions >= 0),
  used_sessions int not null default 0 check (used_sessions >= 0),
  updated_at timestamptz default now()
);

-- ========== توابع ==========
create function handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  v_role text := coalesce(new.raw_user_meta_data->>'role', 'student');
  v_coach uuid;
begin
  if v_role not in ('coach', 'student', 'self') then
    v_role := 'student';
  end if;

  if v_role = 'student' then
    select id into v_coach from profiles
    where invite_code = upper(new.raw_user_meta_data->>'invite_code')
      and role = 'coach';
    if v_coach is null then
      raise exception 'کد دعوت نامعتبر است';
    end if;
  end if;

  insert into profiles (id, username, full_name, role, coach_id, invite_code)
  values (
    new.id,
    lower(new.raw_user_meta_data->>'username'),
    coalesce(new.raw_user_meta_data->>'full_name', ''),
    v_role,
    v_coach,
    case when v_role = 'coach'
      then upper(substr(md5(random()::text), 1, 6)) end
  );
  return new;
end $$;

create trigger on_auth_user_created
after insert on auth.users
for each row execute function handle_new_user();

create function is_my_student(sid uuid) returns boolean
language sql security definer set search_path = public as $$
  select exists (select 1 from profiles where id = sid and coach_id = auth.uid());
$$;

create function is_coach(uid uuid) returns boolean
language sql security definer set search_path = public as $$
  select exists (select 1 from profiles where id = uid and role = 'coach');
$$;

-- ========== فعال‌سازی امنیت سطح ردیف ==========
alter table profiles enable row level security;
alter table exercises enable row level security;
alter table workouts enable row level security;
alter table sets enable row level security;
alter table plan_items enable row level security;
alter table memberships enable row level security;

-- پروفایل: خودت یا شاگردهات
create policy "profiles_select" on profiles for select
  using (id = auth.uid() or coach_id = auth.uid());

-- حرکت‌ها: عمومی، حرکت‌های مربی برای همه، حرکت شخصی فقط برای خودش
create policy "exercises_select" on exercises for select using (
  auth.uid() is not null and
  (created_by is null or created_by = auth.uid() or is_coach(created_by))
);
create policy "exercises_insert" on exercises for insert with check (
  created_by = auth.uid() and
  exists (select 1 from profiles where id = auth.uid() and role in ('coach', 'self'))
);

-- جلسه‌های تمرین
create policy "workouts_select" on workouts for select
  using (student_id = auth.uid() or is_my_student(student_id));
create policy "workouts_insert" on workouts for insert
  with check (student_id = auth.uid());
create policy "workouts_update" on workouts for update
  using (student_id = auth.uid());
create policy "workouts_delete" on workouts for delete
  using (student_id = auth.uid());

-- ست‌ها
create policy "sets_select" on sets for select using (exists (
  select 1 from workouts w
  where w.id = workout_id
    and (w.student_id = auth.uid() or is_my_student(w.student_id))
));
create policy "sets_insert" on sets for insert with check (exists (
  select 1 from workouts w where w.id = workout_id and w.student_id = auth.uid()
));
create policy "sets_delete" on sets for delete using (exists (
  select 1 from workouts w where w.id = workout_id and w.student_id = auth.uid()
));

-- برنامه تمرینی: مربی می‌نویسد، شاگرد می‌خواند
create policy "plan_select" on plan_items for select
  using (student_id = auth.uid() or is_my_student(student_id));
create policy "plan_insert" on plan_items for insert
  with check (is_my_student(student_id));
create policy "plan_update" on plan_items for update
  using (is_my_student(student_id));
create policy "plan_delete" on plan_items for delete
  using (is_my_student(student_id));

-- جلسه‌های باقی‌مانده: مربی تعیین می‌کند، شاگرد فقط می‌بیند
create policy "mem_select" on memberships for select
  using (student_id = auth.uid() or is_my_student(student_id));
create policy "mem_insert" on memberships for insert
  with check (is_my_student(student_id));
create policy "mem_update" on memberships for update
  using (is_my_student(student_id));
